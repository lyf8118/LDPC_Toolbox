/* ========== GPU internal data types for complex numbers ========== */
struct GPU_Complex
{
    float r;
    float i;

    __host__ __device__ GPU_Complex() {}

    // Constructor for a complex number with real and imag parts of float
    __host__ __device__ GPU_Complex(float a, float b) : r(a), i(b) {}

    // Magnitude of a complex number
    __host__ __device__ float magnitude2(void) { return r * r + i * i; }

    __device__ GPU_Complex operator*(const GPU_Complex& a) {
        return GPU_Complex(__fmul_rn(r, a.r) - __fmul_rn(i, a.i), __fmul_rn(i, a.r) + __fmul_rn(r, a.i));
    }

    __device__ GPU_Complex operator*(const float& a) {
        return GPU_Complex(__fmul_rn(r, a), __fmul_rn(i, a));
    }

    __host__ __device__ GPU_Complex operator+(const GPU_Complex& a) {
        return GPU_Complex(r + a.r, i + a.i);
    }

    __host__ __device__ void operator+=(const GPU_Complex& a) {
        r += a.r;
        i += a.i;
    }

    __device__ void multiply_acc(const GPU_Complex& a, const GPU_Complex& b) {
        // real part
        r = __fmaf_rn(a.r, b.r, r);
        r = __fmaf_rn(-a.i, b.i, r);
        // imag part
        i = __fmaf_rn(a.i, b.r, i);
        i = __fmaf_rn(a.r, b.i, i);
    }

    __device__ void multiply_acc(const GPU_Complex& a, const float& b) {
        // real part
        r = __fmaf_rn(a.r, b, r);
        // imag part
        i = __fmaf_rn(a.i, b, i);
    }
};

    }                                                                          \

    codePhaseStep = mxGetPr(prhs[5]);
    earlyLateSpc = mxGetScalar(prhs[6]);
    startIndex = (int*)mxGetData(prhs[7]);

    dataReadFlag = (int)mxGetScalar(prhs[9]);

    {

        fileType = (int)mxGetScalar(prhs[10]);

        d_ppCaCode = (float**)malloc(sizeof(float*) * channelCnt);
        d_ppBasebandSignal = (GPU_Complex**)malloc(sizeof(GPU_Complex*) * channelCnt);
        blksize = chSampleSize[0];
        mwIndex dims[2] = { 0, 0 };
        for (int chInd = 0; chInd < channelCnt; chInd++)
        {
            dims[1] = chInd;
            size_t index = mxCalcSingleSubscript(prhs[1], (size_t)2, dims);
            float* caCode = &caCodeTable[index];
            CHECK(cudaMalloc(&d_ppCaCode[chInd], sizeof(float) * codeLen));
            CHECK(cudaMemcpy(d_ppCaCode[chInd], caCode, codeLen * sizeof(float), cudaMemcpyHostToDevice));
            CHECK(cudaMalloc(&d_ppBasebandSignal[chInd], sizeof(GPU_Complex) * (blksize + 100)));
        }

        blockPerGrid = (int)(blksize + 100 + THREADSPERBLOCK) / THREADSPERBLOCK;

        rawSignalLen = mxGetM(prhs[0]);
        CHECK(cudaMalloc(&d_pRawSignal, rawSignalLen * sizeof(float)));

        CHECK(cudaHostAlloc(&h_corrValues, CORR_NUMBER * channelCnt * sizeof(std::complex<float>), cudaHostAllocMapped));
        CHECK(cudaHostGetDevicePointer(&d_corrValues, h_corrValues, 0));

        for (int chInd = 0; chInd < channelCnt; chInd++) {
            CHECK(cudaStreamCreate(&streams[chInd]));
        }

        mexAtExit(cleanup);
        initialized = 1;
        printf("   CUDA initialized ...\n");
    }
